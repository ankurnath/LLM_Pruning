import numpy as np

def extract_feature(G):
    nodes = list(G.nodes())
    degs = np.fromiter((G.degree[n] for n in nodes), dtype=float, count=len(nodes))
    weights = np.fromiter((G.nodes[n]['weight'] for n in nodes), dtype=float, count=len(nodes))
    result = np.zeros_like(degs, dtype=float)
    np.divide(degs, weights, out=result, where=weights != 0)
    result[degs == 0] = 0.0
    return result

import numpy as np

def extract_feature(G):
    nodes = list(G.nodes())
    weights = np.fromiter((G.nodes[n]['weight'] for n in nodes), dtype=float, count=len(nodes))
    med = float(np.median(weights))
    low = {n for n, w in zip(nodes, weights) if w <= med}
    adj = G.adj
    out = np.empty(len(nodes), dtype=float)
    for i, n in enumerate(nodes):
        nbrs = adj[n]
        deg = len(nbrs)
        if deg == 0:
            out[i] = 0.0
        else:
            c = 0
            for u in nbrs:
                if u in low:
                    c += 1
            out[i] = c / deg
    return out

import numpy as np
import networkx as nx

def extract_feature(G):
    nodes = list(G.nodes())
    deg = dict(G.degree(nodes))
    tri = nx.triangles(G, nodes)
    res = np.empty(len(nodes), dtype=float)
    for i, v in enumerate(nodes):
        d = deg[v]
        res[i] = 0.0 if d < 2 else (2.0 * tri[v]) / (d * (d - 1))
    return res

        n1 = adj[v]

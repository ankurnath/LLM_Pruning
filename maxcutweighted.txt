def extract_feature(G):
    nodelist = list(G.nodes())
    n = len(nodelist)
    if n == 0:
        return np.array([], dtype=float)
    deg = np.array([d for _, d in G.degree(nodelist)], dtype=float)
    alpha = 1.0 / 20.0
    delta = deg - alpha
    pos_delta = delta[delta > 0]
    beta = (n / pos_delta.min()) if pos_delta.size > 0 else 0.0
    w = (beta / n) * delta
    w[w < 1.0] = 1.0
    return w

import numpy as np

def extract_feature(G):
    nodelist = list(G.nodes())
    idx = {n: i for i, n in enumerate(nodelist)}
    neighbor_sets = {n: set(G.neighbors(n)) for n in nodelist}
    deg = {n: len(neighbor_sets[n]) for n in nodelist}
    sums = np.zeros(len(nodelist), dtype=float)
    for u, v in G.edges():
        nu = neighbor_sets[u]
        nv = neighbor_sets[v]
        if len(nu) < len(nv):
            inter = sum(1 for w in nu if w in nv)
        else:
            inter = sum(1 for w in nv if w in nu)
        union_size = deg[u] + deg[v] - inter
        j = (inter / union_size) if union_size > 0 else 0.0
        sums[idx[u]] += j
        sums[idx[v]] += j
    res = np.empty(len(nodelist), dtype=float)
    for n in nodelist:
        i = idx[n]
        d = deg[n]
        res[i] = (sums[i] / d) if d > 0 else 0.0
    return res

import numpy as np

def extract_feature(G):
    nodes = list(G.nodes())
    deg_dict = dict(G.degree())
    deg = np.fromiter((deg_dict[n] for n in nodes), dtype=float)
    if deg.size == 0:
        return deg
    mean = deg.mean()
    std = deg.std()
    if std == 0.0:
        return np.zeros_like(deg)
    return (deg - mean) / std

import numpy as np

def extract_feature(G):
    nodes = list(G.nodes())
    n = len(nodes)
    if n == 0:
        return np.array([], dtype=float)
    deg = np.fromiter((G.degree(v) for v in nodes), dtype=float, count=n)
    mean = deg.mean()
    std = deg.std()
    if std == 0.0:
        return np.zeros(n, dtype=float)
    return (deg - mean) / std

import numpy as np

def extract_feature(G):
    nodes = list(G.nodes())
    idx = {n: i for i, n in enumerate(nodes)}
    if G.is_directed():
        neighbors = {n: set(G.predecessors(n)) | set(G.successors(n)) for n in nodes}
    else:
        neighbors = {n: set(G.neighbors(n)) for n in nodes}
    jaccard_map = {}
    jaccards = []
    for u in nodes:
        Nu = neighbors[u]
        iu = idx[u]
        for v in Nu:
            iv = idx[v]
            if iu < iv:
                Nv = neighbors[v]
                inter = len(Nu & Nv)
                union = len(Nu) + len(Nv) - inter
                j = (inter / union) if union else 0.0
                jaccard_map[(u, v)] = j
                jaccards.append(j)
    tau = float(np.median(jaccards)) if jaccards else 0.0
    result = np.empty(len(nodes), dtype=float)
    for v in nodes:
        iv = idx[v]
        Nv = neighbors[v]
        cnt = 0
        for u in Nv:
            key = (u, v) if idx[u] < iv else (v, u)
            j = jaccard_map.get(key, 0.0)
            if j < tau:
                cnt += 1
        w = G.nodes[v]['weight']
        result[iv] = np.inf if w == 0 else cnt / w
    return result

import numpy as np

def extract_feature(G):
    nodes = list(G.nodes())
    deg_map = dict(G.degree())
    deg_arr = np.array([deg_map[v] for v in nodes], dtype=float)
    means = np.empty(len(nodes), dtype=float)
    adj = G.adj
    for i, v in enumerate(nodes):
        nbrs = adj[v]
        if len(nbrs) == 0:
            means[i] = 0.0
        else:
            s = 0
            for u in nbrs:
                s += deg_map[u]
            means[i] = s / len(nbrs)
    denom = deg_arr + means
    num = deg_arr - means
    ratio = np.zeros_like(deg_arr, dtype=float)
    mask = denom != 0
    ratio[mask] = num[mask] / denom[mask]
    w = np.array([G.nodes[v].get('weight', 0.0) for v in nodes], dtype=float)
    feature = np.zeros_like(ratio)
    wmask = w != 0
    feature[wmask] = ratio[wmask] / w[wmask]
    return feature

import numpy as np

def extract_feature(G):
    nodes = list(G.nodes())
    deg = np.fromiter((G.degree(n) for n in nodes), dtype=float, count=len(nodes))
    if deg.size == 0:
        return deg
    mean_deg = deg.mean()
    std_deg = deg.std()
    if std_deg == 0.0:
        return np.zeros_like(deg, dtype=float)
    return (deg - mean_deg) / std_deg
